function(add_geometry_test target_name source_file)
    add_executable(${target_name} ${source_file})
    target_link_libraries(${target_name} PRIVATE triangles_geometry GTest::gtest_main)
    triangles_enable_warnings(${target_name})

    gtest_discover_tests(${target_name} PROPERTIES TIMEOUT 10 LABELS "geometry")
endfunction()

add_geometry_test(vector3_tests vector3_tests.cpp)
add_geometry_test(triangle_tests triangle_tests.cpp)
add_geometry_test(predicates_tests predicates_tests.cpp)
add_geometry_test(bounding_box_tests bounding_box_tests.cpp)
add_geometry_test(intersection_tests intersection_tests.cpp)
add_geometry_test(input_reader_tests input_reader_tests.cpp)
add_geometry_test(intersection_search_tests intersection_search_tests.cpp)
add_geometry_test(uniform_grid_tests uniform_grid_tests.cpp)
